15class HTMLTextExtractor(html.parser.HTMLParser): 

16 """Simple subclass of :class:`html.parser.HTMLParser`. 

17 

18 Collects data elements (non-tag, -comment, -pi, etc. elements) 

19 fed to the parser, then make them available as stripped, concatenated 

20 text via `HTMLTextExtractor.get_text()`. 

21 

22 Note: 

23 Users probably shouldn't deal with this class directly; 

24 instead, use `:func:`remove.remove_html_tags()`. 

25 """ 

26 

27 def __init__(self): 

28 """Initialize the parser.""" 

29 super().__init__() 

30 self.data = [] 

31 

32 def handle_data(self, data: Any) -> None: 

33 """Handle data elements. 

34 

35 Args: 

36 data (Any): The data element(s) to handle. 

37 """ 

38 self.data.append(data) 

39 

40 def get_text(self, sep: Optional[str] = "") -> str: 

41 """Return the collected text. 

42 

43 Args: 

44 sep (Optional[str]): The separator to join the collected text with. 

45 

46 Returns: 

47 str: The collected text. 

48 """ 

49 return sep.join(self.data).strip()

70RE_URL: Pattern = re.compile( 

71 r"(?:^|(?<![\w/.]))" 

72 # protocol identifier 

73 # r"(?:(?:https?|ftp)://)" <-- alt? 

74 r"(?:(?:https?://|ftp://|www\d{0,3}\.))" 

75 # user:pass authentication 

76 r"(?:\S+(?::\S*)?@)?" r"(?:" 

77 # IP address exclusion 

78 # private & local networks 

79 r"(?!(?:10|127)(?:\.\d{1,3}){3})" 

80 r"(?!(?:169\.254|192\.168)(?:\.\d{1,3}){2})" 

81 r"(?!172\.(?:1[6-9]|2\d|3[0-1])(?:\.\d{1,3}){2})" 

82 # IP address dotted notation octets 

83 # excludes loopback network 0.0.0.0 

84 # excludes reserved space >= 224.0.0.0 

85 # excludes network & broadcast addresses 

86 # (first & last IP address of each class) 

87 r"(?:[1-9]\d?|1\d\d|2[01]\d|22[0-3])" 

88 r"(?:\.(?:1?\d{1,2}|2[0-4]\d|25[0-5])){2}" 

89 r"(?:\.(?:[1-9]\d?|1\d\d|2[0-4]\d|25[0-4]))" 

90 r"|" 

91 # host name 

92 r"(?:(?:[a-z\u00a1-\uffff0-9]-?)*[a-z\u00a1-\uffff0-9]+)" 

93 # domain name 

94 r"(?:\.(?:[a-z\u00a1-\uffff0-9]-?)*[a-z\u00a1-\uffff0-9]+)*" 

95 # TLD identifier 

96 r"(?:\.(?:[a-z\u00a1-\uffff]{2,}))" r")" 

97 # port number 

98 r"(?::\d{2,5})?" 

99 # resource path 

100 r"(?:/\S*)?" r"(?:$|(?![\w?!+&/]))", 

101 flags=re.IGNORECASE, 

102) 

103 

104RE_SHORT_URL: Pattern = re.compile(

106 # optional scheme 

107 r"(?:(?:https?://)?)" 

108 # domain 

109 r"(?:\w-?)*?\w+(?:\.[a-z]{2,12}){1,3}" r"/" 

110 # hash

185@functools.lru_cache(maxsize=None) 

186def _get_punct_translation_table(): 

187 """Get the punctuation translation table.""" 

188 return dict.fromkeys( 

189 ( 

190 i 

191 for i in range(sys.maxunicode) 

192 if unicodedata.category(chr(i)).startswith("P") 

193 ), 

194 " ", 

195 )
