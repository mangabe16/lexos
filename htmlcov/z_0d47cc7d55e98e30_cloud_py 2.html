32def wordcloud( 

33 data: single_doc_types | multi_doc_types | pd.DataFrame, 

34 docs: Optional[int | str | list[int] | list[str]] = None, 

35 opts: Optional[dict[str, Any]] = None, 

36 show: Optional[bool] = True, 

37 figure_opts: Optional[dict[str, Any]] = None,

41 """Make a word cloud. 

42 

43 Accepts data from a string, list of lists or tuples, a dict with 

44 terms as keys and counts/frequencies as values, or a dataframe. 

45 

46 Args: 

47 data (single_doc_types | multi_doc_types): The data. 

48 A single_doc_type can be a dict with terms as keys and counts as values, a Doc, a Span, a string, a list of strings, or a list of Tokens. A multi_doc_type can be a DTM, a DataFrame, a list of Docs, Spans or term-count dicts, or a list of lists of strings or Tokens. Dataframes must have terms as the index and documents as the columns. 

49 docs (Optional[int | str | list[int] | list[str]]): A list of documents to be selected from the DTM. Values can be either indices or labels. 

50 opts (Optional[dict[str, Any]): The WordCloud() options. 

51 For testing, try {"background_color": "white", "max_words": 2000, "contour_width": 3, "contour_color": "steelblue"} 

52 show (Optional[bool]): Whether to show the plotted word cloud or return it as a WordCloud object. 

53 figure_opts (Optional[dict[str, Any]]): A dict of matplotlib figure options. 

54 round (Optional[int]): An integer (generally between 100-300) to apply a mask that rounds the word cloud. 

55 path (Optional[Path | str]): The filepath to save the word cloud to. 

56 

57 Returns: 

58 WordCloud: A WordCloud object if show is set to False. 

59 

60 Notes: 

61 - For a full list of options, see https://amueller.github.io/word_cloud/generated/wordcloud.WordCloud.html#wordcloud-wordcloud. 

62 - If `show=False` the function expects to be called with something like `wordcloud = make_wordcloud(data, show=False)`. 

63 This returns WordCloud object which can be manipulated by any of its methods, such as `to_file()`. See the 

64 WordCloud documentation for a list of methods. 

65 """ 

66 # Set the default options 

67 if opts is None: 

68 opts = { 

69 "background_color": "white", 

70 "max_words": 2000, 

71 "contour_width": 0, 

72 "contour_color": "steelblue", 

73 } 

74 if figure_opts is None: 

75 figure_opts = {} 

76 

77 # Set the mask, if using 

78 if round: 

79 x, y = np.ogrid[:300, :300] 

80 mask = (x - 150) ** 2 + (y - 150) ** 2 > round**2 

81 mask = 255 * mask.astype(int) 

82 opts["mask"] = mask 

83 

84 # Process the data into a consistent format 

85 if isinstance(data, str): 

86 wc = WordCloud(**opts).generate_from_text(data) 

87 elif isinstance(data, (Doc | Span)): 

88 wc = WordCloud(**opts).generate_from_frequencies( 

89 Counter([t.text for t in data]) 

90 ) 

91 else: 

92 if isinstance(data, DTM): 

93 counts = processors.process_dtm(data, docs) 

94 elif isinstance(data, pd.DataFrame): 

95 counts = processors.process_dataframe(data, docs) 

96 elif isinstance(data, list) and isinstance(data[0], list): 

97 counts = processors.process_list(data, docs) 

98 elif isinstance(data, list) and isinstance(data[0], (Doc | Span)): 

99 counts = processors.process_docs(data, docs) 

100 elif isinstance(data, list) and not isinstance(data[0], list): 

101 counts = processors.process_item(data) 

102 elif isinstance(data, dict): 

103 counts = data 

104 else: 

105 raise LexosException( 

106 "Cannot process data. Make sure that all items in the input belong to the same data type." 

107 ) 

108 wc = WordCloud(**opts).generate_from_frequencies(counts) 

109 

110 # Plot or return the WordCloud 

111 if show: 

112 if figure_opts: 

113 plt.figure(**figure_opts) 

114 plt.axis("off") 

115 # If a filepath is provided, save the figure 

116 if path: 

117 wc.to_file(path) 

118 plt.imshow(wc) 

119 plt.show() 

120 else: 

121 return wc

142 

143 Accepts data from a string, list of lists, a dict with

145 

146 The best input is a dtm produced by `get_dtm_table()`. 

147

149 data (multi_doc_types): The data. 

150 A multi_doc_type can be a DTM, a DataFrame, a list of Docs, Spans or term-count dicts, or a list of lists of strings or Tokens. Dataframes must have terms as the index and documents as the columns. 

151 as values, or a dataframe with "term" and "count" columns. 

152 docs: (Optional[int | str | list[int] | list[str]]): A list of documents to be selected from the DTM. 

153 opts (Optional[dict[str, Any]]): The WordCloud() options. 

154 For testing, try {"background_color": "white", "max_words": 2000, "contour_width": 3, "contour_color": "steelblue"} 

155 ncols (Optional[int]): The number of columns in the grid. 

156 title (Optional[str]): The title of the grid. 

157 labels (Optional[list[str]]): The document labels for each subplot. 

158 show (Optional[bool]): Whether to show the plotted word cloud or return it as a WordCloud object. 

159 figure_opts (Optional[dict[str, Any]]): A dict of matplotlib figure options. 

160 round (Optional[int]): An integer (generally between 100-300) to apply a mask that rounds the word cloud. 

161 filename (Optional[str]): The filename to save the figure to. 

162 

163 Returns: 

164 list[WordCloud]: A WordCloud object if show is set to False. 

165 

166 Notes: 

167 - For a full list of options, see https://amueller.github.io/word_cloud/generated/wordcloud.WordCloud.html#wordcloud-wordcloud. 

168 - If `show=False` the function expects to be called with something like `wordcloud = make_wordcloud(data, show=False)`. 

169 This returns WordCloud object which can be manipulated by any of its methods, such as `to_file()`. See the 

170 WordCloud documentation for a list of methods. 

171 """ 

172 if opts is None: 

173 opts = { 

174 "background_color": "white", 

175 "max_words": 2000, 

176 "contour_width": 0,

180 figure_opts = {}
